  if (sbChannel) { try { client.removeChannel(sbChannel); } catch(e) {} sbChannel = null; }
  sbChannel = client.channel('cuidarbem-' + familyCode, { config: { broadcast: { self: false } } })

<script id="cuidarbem-v39-delete-modal-script">
let cbDeleteConfirmCallback = null;
function openDeleteConfirm(opts){
  const o = opts || {};
  cbDeleteConfirmCallback = typeof o.onConfirm === 'function' ? o.onConfirm : null;
  const overlay = document.getElementById('cb-delete-overlay');
  const title = document.getElementById('cb-delete-title');
  const msg = document.getElementById('cb-delete-message');
  const btn = document.getElementById('cb-delete-confirm-btn');
  if (!overlay || !title || !msg || !btn) {
    if (cbDeleteConfirmCallback) cbDeleteConfirmCallback();
    return;
  }
  title.textContent = o.title || 'Excluir?';
  msg.textContent = o.message || 'Esta ação não pode ser desfeita.';
  btn.textContent = o.confirmText || 'Excluir';
  btn.onclick = function(){
    const fn = cbDeleteConfirmCallback;
    cbDeleteConfirmCallback = null;
    overlay.classList.remove('open');
    if (fn) fn();
  };
  overlay.classList.add('open');
  setTimeout(()=>btn.focus && btn.focus(), 80);
}
function closeDeleteConfirm(ev){
  if (ev && ev.target && ev.target.id !== 'cb-delete-overlay') return;
  cbDeleteConfirmCallback = null;
  const overlay = document.getElementById('cb-delete-overlay');
  if (overlay) overlay.classList.remove('open');
}
document.addEventListener('keydown', function(e){
  if (e.key === 'Escape') closeDeleteConfirm();
});
</script>
