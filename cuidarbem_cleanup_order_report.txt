<script id="cb-wheel-router">
/* Desktop: rota wheel para o container scrollável sob o cursor.
   Home screen: grid com overflow:hidden — detecta painel por posição X do cursor. */
(function(){
  if (typeof window === 'undefined') return;

  function getScrollParent(el) {
    while (el && el !== document.documentElement) {
      var st = window.getComputedStyle(el);
      var oy = st.overflowY;
      if ((oy === 'auto' || oy === 'scroll') && el.scrollHeight > el.clientHeight) {
        return el;
      }
      el = el.parentElement;
    }
    return null;
  }

  document.addEventListener('wheel', function(e) {
    if (window.innerWidth < 768) return;

    // Tela Início: grid com overflow:hidden — getScrollParent falha.
    // Roteia pelo painel sob o cursor (esquerda=content, direita=right-panel).
    var home = document.getElementById('screen-home');
    if (home && home.classList.contains('active')) {
      var rightPanel = home.querySelector('.desktop-right-panel');
      var panel = (rightPanel && e.clientX >= rightPanel.getBoundingClientRect().left)
        ? rightPanel : home.querySelector('.content');
      if (panel && panel.scrollHeight > panel.clientHeight) {
        e.preventDefault();
        panel.scrollBy({ top: e.deltaY * (e.deltaMode === 1 ? 20 : 1), behavior: 'auto' });
      }
      return; // sempre consome o evento na home
    }

    // Outras telas: sobe a árvore até o scroll container natural
    var target = getScrollParent(e.target);
    if (!target) return;

    var atTop    = target.scrollTop <= 0;
    var atBottom = target.scrollTop + target.clientHeight >= target.scrollHeight - 1;
    if ((e.deltaY < 0 && atTop) || (e.deltaY > 0 && atBottom)) return;

    e.preventDefault();
    target.scrollBy({ top: e.deltaY * (e.deltaMode === 1 ? 20 : 1), behavior: 'auto' });
  }, { passive: false });
})();
</script>
